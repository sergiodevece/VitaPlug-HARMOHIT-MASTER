# R12 DRIVE audit harness. Isolated: does not touch product targets.
# Configure: cmake -S Documentation/R12-audit/tools -B <scratch>/r12 -DCMAKE_BUILD_TYPE=Release
cmake_minimum_required(VERSION 3.22)
project(HarmoHitR12DriveAudit LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(MASTER_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
set(MIX_ROOT "${MASTER_ROOT}/../harmohit" CACHE PATH "HARMOHIT MIX checkout (read only)")
set(JUCE_DIR "${MASTER_ROOT}/../vitaplug_gainer_80/external/JUCE" CACHE PATH "Pinned JUCE 9.0.3")
add_subdirectory("${JUCE_DIR}" JUCE)
add_executable(r12_drive_audit audit_drive.cpp)
target_include_directories(r12_drive_audit PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/r1_frozen" "${MIX_ROOT}/Source/DSP" "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(r12_drive_audit PRIVATE JUCE_STANDALONE_APPLICATION=1 JUCE_USE_CURL=0 JUCE_WEB_BROWSER=0)
target_link_libraries(r12_drive_audit PRIVATE juce::juce_dsp juce::juce_audio_basics juce::juce_recommended_config_flags)
add_executable(wav_ab wav_ab.cpp)
add_executable(r12_compare r12_compare.cpp)
target_include_directories(r12_compare PRIVATE "${MASTER_ROOT}/Source/DSP" "${MIX_ROOT}/Source/DSP" "${CMAKE_CURRENT_SOURCE_DIR}/r1_frozen")
target_compile_definitions(r12_compare PRIVATE JUCE_STANDALONE_APPLICATION=1 JUCE_USE_CURL=0 JUCE_WEB_BROWSER=0)
target_link_libraries(r12_compare PRIVATE juce::juce_dsp juce::juce_audio_basics juce::juce_recommended_config_flags)
add_executable(r12_bench bench.cpp)
target_include_directories(r12_bench PRIVATE "${MASTER_ROOT}/Source/DSP" "${CMAKE_CURRENT_SOURCE_DIR}/r1_frozen")
target_compile_definitions(r12_bench PRIVATE JUCE_STANDALONE_APPLICATION=1 JUCE_USE_CURL=0 JUCE_WEB_BROWSER=0)
target_link_libraries(r12_bench PRIVATE juce::juce_dsp juce::juce_audio_basics juce::juce_recommended_config_flags)
